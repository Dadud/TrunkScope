cmake_minimum_required(VERSION 3.24)
project(trunkscope-radiod VERSION 0.1.0 LANGUAGES CXX)
option(TRUNKSCOPE_WITH_SOAPY "Build the SoapySDR receiver backend" OFF)
option(TRUNKSCOPE_WITH_OP25 "Build the OP25/GNU Radio decoder backend" OFF)
add_executable(trunkscope-radiod src/main.cpp)
target_compile_features(trunkscope-radiod PRIVATE cxx_std_20)
if(MSVC)
  target_compile_options(trunkscope-radiod PRIVATE /W4 /permissive-)
else()
  target_compile_options(trunkscope-radiod PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(TRUNKSCOPE_WITH_SOAPY)
  find_package(SoapySDR CONFIG REQUIRED)
  target_link_libraries(trunkscope-radiod PRIVATE SoapySDR)
  target_compile_definitions(trunkscope-radiod PRIVATE TRUNKSCOPE_WITH_SOAPY=1)
endif()
install(TARGETS trunkscope-radiod RUNTIME DESTINATION bin)
